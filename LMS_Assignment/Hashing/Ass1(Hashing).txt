package com.hashing;

import java.util.Arrays;

public class HashingArray {

	private int[] hashtable;
	private int size;

	public int hashkey(int val) {
		return val % size;
	}

	public void createhash(int size) {
		this.size = size;
		this.hashtable = new int[size];
		for (int i = 0; i < size; i++) {
			hashtable[i] = -1;
		}
	}

	public void addKey(int val) {
		int key = hashkey(val);
		if (hashtable[key] == -1) {
			hashtable[key] = val;
		} else {
			int i;
			for (i = key + 1; i != key && hashtable[i] != -1; i = (i + 1) % size)
				;
			hashtable[i] = key;
		}
	}

	public void search(int val) {
		int key = hashkey(val);

		if (hashtable[key] == val) {
			hashtable[key] = key;
		} else {
			int i;
			for (i = key + 1; i != key && hashtable[i] != key; i = (i + 1) % size)
				;
			if (hashtable[i] == key) {
				System.out.println("Element is found" + i);
			}
			System.out.println("Element is not found");
		}
	}

	public void display() {
		System.out.println(Arrays.toString(hashtable));
//			System.out.println(" " + hashtable[i]);
	}
}

package com.hashing;

import java.io.BufferedReader;
import java.io.InputStreamReader;

public class HashPack {

	public static void main(String[] args) throws Exception {

		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

		System.out.print("Enter size of array : ");
		int size = Integer.parseInt(br.readLine());

		HashingArray ha = new HashingArray();

		ha.createhash(size);

		while (true) {
			System.out.println("\n1.Add" + "\n2.Search" + "\n3.display" + "\n4.Delete" + "\n5.Exit");

			System.out.print("Enter your choice : ");
			int choice = Integer.parseInt(br.readLine());

			switch (choice) {
			case 1:
				System.out.print("Enter value : ");
				int value = Integer.parseInt(br.readLine());
				ha.addKey(value);
				break;

			case 2:
				System.out.print("Enter search value : ");
				value = Integer.parseInt(br.readLine());
				ha.search(value);
				break;

			case 3:
				System.out.print("Hash Array : ");
				ha.display();
				break;

			case 5:
				System.out.println("Code is terminated...");
				System.exit(0);

			default:
				throw new IllegalArgumentException("Unexpected value: " + choice);
			}
		}

	}

}
